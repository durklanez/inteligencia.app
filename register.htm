async function login() { const email = document.getElementById("loginEmail").value.trim(); const senha = document.getElementById("loginSenha").value; const msg = document.getElementById("loginMsg"); if (!email || !senha) { msg.className = "error"; msg.innerText = "Preencha email e senha."; return; } try { const resposta = await fetch("/login", { method: "POST", headers: { "Content-Type": "application/json", "Accept": "application/json" }, body: JSON.stringify({ email: email, senha: senha }) }); const texto = await resposta.text(); let dados; try { dados = JSON.parse(texto); } catch (e) { console.error( "Resposta do servidor:", texto ); throw new Error( "O servidor não respondeu corretamente." ); } if (!resposta.ok) { msg.className = "error"; msg.innerText = dados.erro || dados.error || "Login inválido."; return; } console.log( "LOGIN FRONTEND OK:", dados ); if (!dados.usuario) { console.error( "Resposta sem usuario:", dados ); throw new Error( "O servidor não enviou os dados do usuário." ); } localStorage.setItem( "user", JSON.stringify(dados.usuario) ); entrarNoMenu(dados.usuario); } catch (erro) { console.error( "Erro no login:", erro ); msg.className = "error"; msg.innerText = erro.message || "Erro de conexão com o servidor."; } }
